27impl VolumeElementTypes {
28    pub const NUMBER_OF_TYPES: usize = 8; //G and non G type count for 1
29
30    pub fn to_index(&self) -> usize {
31        match self {
32            VolumeElementTypes::Tetra4 | VolumeElementTypes::GTetra4 => 0,
33            VolumeElementTypes::Tetra10 | VolumeElementTypes::GTetra10 => 1,
34            VolumeElementTypes::Pyramid5 | VolumeElementTypes::GPyramid5 => 2,
35            VolumeElementTypes::Penta6 | VolumeElementTypes::GPenta6 => 3,
36            VolumeElementTypes::Penta15 | VolumeElementTypes::GPenta15 => 4,
37            VolumeElementTypes::Hexa8 | VolumeElementTypes::GHexa8 => 5,
38            VolumeElementTypes::Hexa20 | VolumeElementTypes::GHexa20 => 6,
39            VolumeElementTypes::GPyramid13 | VolumeElementTypes::Pyramid13 => 7,
40        }
41    }
42}

44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub enum ElementsType {
46    Point,
47    GPoint,
48    Bar2,
49    GBar2,
50    Bar3,
51    GBar3,
52    Tria3,
53    GTria3,
54    Tria6,
55    GTria6,
56    Quad4,
57    GQuad4,
58    Quad8,
59    GQuad8,

73impl FromStr for ElementsType {
74    type Err = ();
75
76    fn from_str(s: &str) -> Result<Self, Self::Err> {
77        use ElementsType::*;
78        use VolumeElementTypes::*;
79
80        match s.trim().to_lowercase().as_str() {
81            "point" => Ok(Point),
82            "g_point" => Ok(GPoint),
83            "bar2" => Ok(Bar2),
84            "g_bar2" => Ok(GBar2),
85            "bar3" => Ok(Bar3),
86            "g_bar3" => Ok(GBar3),
87            "tria3" => Ok(Tria3),
88            "g_tria3" => Ok(GTria3),
89            "tria6" => Ok(Tria6),
90            "g_tria6" => Ok(GTria6),
91            "quad4" => Ok(Quad4),
92            "g_quad4" => Ok(GQuad4),
93            "quad8" => Ok(Quad8),
94            "g_quad8" => Ok(GQuad8),
95            "tetra4" => Ok(VolumeElementType(Tetra4)),
96            "g_tetra4" => Ok(VolumeElementType(GTetra4)),
97            "tetra10" => Ok(VolumeElementType(Tetra10)),
98            "g_tetra10" => Ok(VolumeElementType(GTetra10)),
99            "pyramid5" => Ok(VolumeElementType(Pyramid5)),
100            "g_pyramid5" => Ok(VolumeElementType(GPyramid5)),
101            "pyramid13" => Ok(VolumeElementType(Pyramid13)),
102            "g_pyramid13" => Ok(VolumeElementType(GPyramid13)),
103            "penta6" => Ok(VolumeElementType(Penta6)),
104            "g_penta6" => Ok(VolumeElementType(GPenta6)),
105            "penta15" => Ok(VolumeElementType(Penta15)),
106            "g_penta15" => Ok(VolumeElementType(GPenta15)),
107            "hexa8" => Ok(VolumeElementType(Hexa8)),
108            "g_hexa8" => Ok(VolumeElementType(GHexa8)),
109            "hexa20" => Ok(VolumeElementType(Hexa20)),
110            "g_hexa20" => Ok(VolumeElementType(GHexa20)),
111            "nsided" => Ok(Nsided),
112            "g_nsided" => Ok(GNsided),
113            "nfaced" => Ok(Nfaced),
114            "g_nfaced" => Ok(GNfaced),
115            _ => Err(()),
116        }
117    }
118}

120impl ElementsType {
121    // Consolidated method to get node count and vertex count, assuming they are the same.
122    pub fn node_count(&self) -> i32 {
123        use ElementsType::*;
124        use VolumeElementTypes::*;
125
126        match self {
127            Point | GPoint => 1,
128            Bar2 | GBar2 => 2,
129            Bar3 | GBar3 => 3,
130            Tria3 | GTria3 => 3,
131            Tria6 | GTria6 => 6,
132            Quad4 | GQuad4 => 4,
133            Quad8 | GQuad8 => 8,
134
135            Nsided | GNsided => panic!("Nsided has variable node count"),
136            Nfaced | GNfaced => -1,
137            VolumeElementType(Tetra4) | VolumeElementType(GTetra4) => 4,
138            VolumeElementType(Tetra10) | VolumeElementType(GTetra10) => 10,
139            VolumeElementType(Pyramid5) | VolumeElementType(GPyramid5) => 5,
140            VolumeElementType(Penta6) | VolumeElementType(GPenta6) => 6,
141            VolumeElementType(Penta15) | VolumeElementType(GPenta15) => 15,
142            VolumeElementType(Hexa8) | VolumeElementType(GHexa8) => 8,
143            VolumeElementType(Hexa20) | VolumeElementType(GHexa20) => 20,
144            VolumeElementType(Pyramid13) | VolumeElementType(GPyramid13) => 13,
145        }
146    }
