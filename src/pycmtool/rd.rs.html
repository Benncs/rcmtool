18#[pyclass(name = "RawDataScalar", frozen)]
19pub struct RawDataScalarWrapper(pub(crate) cmtool_data::RawDataScalar);
20#[pymethods]
21impl RawDataScalarWrapper {
22    #[getter]
23    pub fn n_zone(&self) -> usize {
24        self.0.header.n_zone as usize
25    }
26
27    #[getter]
28    pub fn data(&self, py: Python<'_>) -> Py<PyArray1<f64>> {
29        let values: Vec<f64> = self.0.values.iter().map(|f| f.value).collect();
30        //from does not perform copy
31        let array = numpy::ndarray::Array1::from(values);
32        PyArray1::from_owned_array(py, array).unbind()
33    }
34
35    pub fn write(&self, path: &str) -> PyResult<()> {
36        self.0.write_raw(path).map_err(PythonError::from)?;
37        Ok(())
38    }
39}

41#[pyfunction]
42pub fn read_rawscalar(path: &str) -> PyResult<RawDataScalarWrapper> {
43    let scalar =
44        cmtool_data::RawDataScalar::read_raw(path).ok_or(PythonError(DataError::BadData))?;
45    Ok(RawDataScalarWrapper(scalar))
46}

48#[pyfunction]
49pub fn scalar_from_data<'py>(
50    _py: Python<'py>,
51    x: numpy::PyReadonlyArrayDyn<'py, f64>,
52) -> PyResult<RawDataScalarWrapper> {
53    if x.shape().len() != 1 {
54        return Err(PyValueError::new_err("Input array must be 1D."));
55    }
56
57    if !x.is_contiguous() {
58        return Err(PyValueError::new_err(
59            "Input array must be contiguous in memory.",
60        ));
61    }
62
63    match x.as_slice() {
64        Ok(slice) => {
65            //TODO Why this condition has been used ?
66            // if slice.len() != 1 {
67            //     return Err(PyValueError::new_err(
68            //         "Input array must contain exactly one element.",
69            //     ));
70            // }
71            let scalar = RawDataScalar::from(slice);
72            Ok(RawDataScalarWrapper(scalar))
73        }
74        Err(_) => Err(PyValueError::new_err(
75            "Failed to convert the array to a contiguous slice.",
76        )),
77    }
78}

150#[pyclass(name = "FlowMapDescriptor")]
151pub struct FlowMapDescriptorWrapper(cmtool_data::FlowMapDescriptor);
152#[pymethods]
153impl FlowMapDescriptorWrapper {
154    //TODO check safety of this, maybe use RC<refcell> to do not have rust mutability
155    #[getter]
156    pub fn flowmap(this: Bound<'_, Self>) -> Bound<'_, PyArray2<f64>> {
157        let flowmap = &this.borrow().0.flowmap;
158        // SAFETY:
159        // - The returned NumPy array shares memory with the internal `flowmap` (Array2<f64>).
160        // - We use `borrow_from_array`, which ties the array's lifetime to the Python object (`this`).
161        // - This guarantees that the underlying Rust memory remains valid as long as Python holds the array.
162        //
163        // Critical Requirements:
164        // - `self.0.flowmap` must not be mutated in a way that causes memory reallocation (e.g., replacing it).
165        //   For example, the following code is unsafe if it runs after `pyobject.flowmap` is accessed:
166        //
167        //     fn drop(&mut self) {
168        //         self.0.flowmap = Array2::zeros((1, 1));  // BAD: reallocates backing buffer
169        //     }
170        //
171        // - Violating this invariant (e.g., replacing the array or shrinking it) while Python holds a reference
172        //   will cause undefined behavior (likely a segmentation fault).
173        //
174        // - Only expose immutable views or ensure exclusive access if mutations are needed.
175        unsafe { PyArray2::borrow_from_array(flowmap, this.into_any()) }
176    }
177
178    #[getter]
179    pub fn neighbors(this: Bound<'_, Self>) -> Bound<'_, PyArray2<usize>> {
180        let flowmap = &this.borrow().0.neighbors;
181        // SAFETY:
182        // - The returned NumPy array shares memory with the internal `flowmap` (Array2<f64>).
183        // - We use `borrow_from_array`, which ties the array's lifetime to the Python object (`this`).
184        // - This guarantees that the underlying Rust memory remains valid as long as Python holds the array.
185        //
186        // Critical Requirements:
187        // - `self.0.flowmap` must not be mutated in a way that causes memory reallocation (e.g., replacing it).
188        //   For example, the following code is unsafe if it runs after `pyobject.flowmap` is accessed:
189        //
190        //     fn drop(&mut self) {
191        //         self.0.flowmap = Array2::zeros((1, 1));  // BAD: reallocates backing buffer
192        //     }
193        //
194        // - Violating this invariant (e.g., replacing the array or shrinking it) while Python holds a reference
195        //   will cause undefined behavior (likely a segmentation fault).
196        //
197        // - Only expose immutable views or ensure exclusive access if mutations are needed.
198        unsafe { PyArray2::borrow_from_array(flowmap, this.into_any()) }
199    }
200
201    #[getter]
202    pub fn volumes(&self, py: Python<'_>) -> Py<PyArray1<f64>> {
203        let array = numpy::ndarray::Array1::from(self.0.volumes.clone());
204        PyArray1::from_owned_array(py, array).unbind()
205    }
206}

208#[pyfunction]
209pub fn read_flowmap(
210    _py: Python<'_>,
211    path: &str,
212    path_2: &str,
213) -> PyResult<FlowMapDescriptorWrapper> {
214    let f = cmtool_data::RawDataFlux::read_raw(path).ok_or(PythonError(DataError::BadData))?;
215    let v = cmtool_data::RawDataScalar::read_raw(path_2).ok_or(PythonError(DataError::BadData))?;
216    let fm = cmtool_data::FlowMapDescriptor::from_raw_data(&f, &v).map_err(PythonError::from)?;
217    Ok(FlowMapDescriptorWrapper(fm))
