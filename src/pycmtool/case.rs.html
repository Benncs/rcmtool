8//Use C compartible enum because Enum-Struct not supported by PyO3
9// #[pyclass(name = "CMExportType")]
10#[pyclass(from_py_object, name = "CMExportType")]
11#[derive(Clone, Copy)]
12pub enum CMExportTypeWrapper {
13    LiquidFlow,
14    GasFlow,
15    GasVolume,
16    LiquidVolume,
17    EnergyDissipation,
18    Kla,
19    Other,
20}
21
22impl From<CMExportTypeWrapper> for cmtool_data::CMAExportType {
23    fn from(value: CMExportTypeWrapper) -> Self {
24        match value {
25            CMExportTypeWrapper::LiquidFlow => Self::LiquidFlow,
26            CMExportTypeWrapper::GasFlow => Self::GasFlow,
27            CMExportTypeWrapper::LiquidVolume => Self::LiquidVolume,
28            CMExportTypeWrapper::GasVolume => Self::GasVolume,
29            CMExportTypeWrapper::EnergyDissipation => Self::EnergyDissipation,
30            CMExportTypeWrapper::Kla => Self::Kla,
31            CMExportTypeWrapper::Other => Self::Other,
32        }
33    }
34}

36// #[pyclass(name = "CMCase")]
37#[pyclass(from_py_object, name = "CMCase")]
38#[derive(Clone)]
39pub struct CMCaseWrapper(cmtool_data::CMCase);
40
41#[pymethods]
42impl CMCaseWrapper {
43    pub fn add(&mut self, stype: CMExportTypeWrapper, relative_path: &str) {
44        let c: cmtool_data::CMAExportType = stype.into();
45        self.0.add(c, relative_path);
46    }
47
48    pub fn write(&self, _py: Python<'_>, path: &str) -> PyResult<()> {
49        let p = std::path::Path::new(path);
50        let c = self.0.clone();
51        cmtool_data::CMCaseJson::write_case(c, p).map_err(PythonError::from)?;
52        Ok(())
53    }
54    pub fn resolve(&self, root: &str, stype: CMExportTypeWrapper) -> Option<Vec<String>> {
55        self.0.resolve_all(root, stype.into())
56    }
57}

59#[pyfunction]
60pub fn make_cm_case(
61    _py: Python<'_>,
62    n_div: [u32; 3],
63    time_per_flow_map: f64,
64    description: Option<String>,
65    recursive: bool,
66) -> CMCaseWrapper {
67    CMCaseWrapper(cmtool_data::CMCase::new(
68        n_div,
69        time_per_flow_map,
70        description,
71        recursive,
72    ))
73}
74
75fn read_cm_case_gen(path: &str) -> PyResult<CMCaseWrapper> {
76    let p = std::path::Path::new(path);
77    let ejson = cmtool_data::read_case(p);
78    let c = ejson.map_err(PythonError::from)?;
79    Ok(CMCaseWrapper(c))
80}

82#[pyfunction]
83pub fn read_cm_case(_py: Python<'_>, path: &str) -> PyResult<CMCaseWrapper> {
84    read_cm_case_gen(path)
